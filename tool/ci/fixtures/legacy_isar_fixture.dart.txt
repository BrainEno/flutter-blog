// Copied into a checkout of the pre-migration app and run with Isar 3.1.0+1.
import 'dart:io';

import 'package:belog/core/theme/theme_preference.dart';
import 'package:belog/features/blog/domain/entities/blog.dart';
import 'package:belog/features/blog/domain/entities/blog_draft.dart';
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:isar/isar.dart';

void main() {
  test('write or verify the legacy database', () async {
    final directory = Platform.environment['ISAR_FIXTURE_DIR']!;
    final mode = Platform.environment['ISAR_FIXTURE_MODE']!;
    await Isar.initializeIsarCore(download: true);
    final isar = await Isar.open(
      [ThemePreferenceSchema, BlogSchema, BlogDraftSchema],
      directory: directory,
      inspector: false,
    );
    try {
      if (mode == 'write') {
        await isar.writeTxn(() async {
          await isar.themePreferences.put(
            ThemePreference(themeMode: ThemeMode.light)..id = 2,
          );
          await isar.themePreferences.put(
            ThemePreference(themeMode: ThemeMode.dark)..id = 9,
          );
          await isar.blogDrafts.put(BlogDraft(
            blogId: 'article-1',
            title: 'legacy draft',
            content: 'unsent content',
            tags: ['legacy'],
            updatedAt: DateTime.utc(2024, 1, 1),
          )..id = 24);
          await isar.blogs.put(Blog(
            iid: 31,
            id: 'cached-1',
            title: 'cached article',
            content: 'cached body',
            posterId: 'author-1',
            imageUrl: 'https://example.invalid/image',
            tags: ['cached'],
            updatedAt: DateTime.utc(2024, 1, 2),
            posterName: 'Author',
            posterAvatar: 'https://example.invalid/avatar',
          ));
        });
      } else if (mode == 'verify') {
        expect((await isar.themePreferences.get(0))?.themeMode, ThemeMode.light);
        final draft = await isar.blogDrafts.get(24);
        expect(draft?.title, 'community draft');
        expect(draft?.content, 'unsent content');
        expect((await isar.blogs.get(31))?.posterName, 'Author');
      } else {
        fail('Unknown fixture mode: $mode');
      }
    } finally {
      await isar.close();
    }
  });
}
